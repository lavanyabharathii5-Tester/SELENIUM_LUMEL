import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver:

public class AddTodoItem {
public static void main(String[] args)
{
webdriver driver = new chromeDriver();

driver.get("https://todomvc.com/examples/angular/dist/browser/#/all");
driver.manage().window().maximize();

WebDriverWait wait = new WebDriverWait{driver.Duration.ofSeconds(10));

//1. Added the items to list
WebElement input = wait.until(ExpectedConditions.visibilityOfElementLocated(By.xpath("//input[@placeholder='What needs to be done?']));
String task = "Learn Selenium"
input.sendkeys(task);

//2. Mark the items as completed.
Webelement checkbox= wait.until(ExpectedConditions.elementToBeClickable(By.Xpath("//a[@routerlink='/completed']"));


//3.Double clicking on a to-do item allows edit.

WebElement item= wait.until(ExpectedConditions.visibilityOfElementLocated(By.xpath(""//label[text()='Learn Selenium'"));

Actions actions = new Actions(driver);
actions.doubleClick().perfrom();

//4. Remove the item
Actions(driver).moveToElement(item).perform();

Webelement cancel = wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//li[.//label[text()='Learn Selenium')]"+"//button[contains(@class,'destroy')]")));

//Verify the remaining item count

WebElement count = wait.until(ExpectedConditions.visibilityOfElementLocated(By.xpath("//span[@class='todo-count']")));
System.out.println("Remaining:"+count.getText());

//select the Active Filter 

driver.findelement(By.xpath("//a[normalize-space()='Active']")).click();

System.out.println("Active items:"+driver.findelements(By.xpath("//ul[@class='todo-list']/li")).size());

//select all filter 

driver.findelement(By.xpath("//a[normalize-space()='All']")).click();

System.out.println("All items:"+driver.findlements(By.xpath("//ul[@class='todo-list']/li")).size());

driver.quit();
}
}
